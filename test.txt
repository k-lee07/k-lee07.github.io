    def stopping_condition(self, node: Node) -> bool:

        :return: True if stopping condition is met, False otherwise
        """

    def gini(self, X: pd.DataFrame, y: pd.Series, feature: str) -> float:
        """
        Returns gini index of the give feature

        :param X: data
        :param y: labels
        :param feature: name of the feature you want to use to get gini score
        :return:
        """
        total_labels = len(y)
        gini_index = 0.0

        #get all the unique y values for feature
        for y_value in np.unique(X[feature]):
            y_subset = y[X[feature] == y_value]
            y_subset_size = len(y_subset)

            if y_subset_size == 0:
                continue
            
            probability = y_subset.value_counts(normalize=True)

            subset_gini = 1 - np.sum(probability ** 2)

            gini_index += (y_subset_size/total_labels) * subset_gini

        return gini_index

    def entropy(self, X: pd.DataFrame, y: pd.Series, feature: str) ->float:
        """
        Returns entropy of the give feature

        :param X: data
        :param y: labels
        :param feature: name of the feature you want to use to get entropy score
        :return:
        """      
        total_labels = len(y)
        entropy = 0.0

        
        for data in np.unique(X[feature]):
            y_subset = y[X[feature] == data]
            subset_size = len(y_subset)

            if subset_size == 0:
                continue
            
            # Step 1: calculate probability of each class
            probability = y_subset.value_counts(normalize=True)

            #Step 2: entropy formula
            subset_entropy = -np.sum(probability * np.log2(probability))

            entropy += (subset_size/total_labels) * subset_entropy

        return entropy
